docker --version
mkdir -p ~/flask-docker
cd ~/flask-docker

nano app.py
  from flask import Flask

  app = Flask(__name__)
  
  # Page 1: Products page
  @app.route('/')
  def products():
      return '''
      <h1>My Online Shop</h1>
      <h2>Products</h2>
      <ul>
          <li>Headphones - Rs. 1999</li>
          <li>Backpack - Rs. 1299</li>
          <li>Water Bottle - Rs. 399</li>
      </ul>
      <a href="/cart">Go to Cart</a>
      '''
  
  # Page 2: Cart page
  @app.route('/cart')
  def cart():
      return '''
      <h1>My Online Shop</h1>
      <h2>Your Cart</h2>
      <ul>
          <li>Headphones - Rs. 1999</li>
          <li>Water Bottle - Rs. 399</li>
      </ul>
      <p>Total: Rs. 2398</p>
      <a href="/">Back to Products</a>
      '''
  
  if __name__ == '__main__':
      app.run(host='0.0.0.0', port=5000)


nano Dockerfile
      FROM python:3.12-slim
      WORKDIR /app
      RUN pip install Flask
      COPY . .
      ENV FLASK_APP=app.py
      EXPOSE 5000
      CMD ["flask", "run", "--host=0.0.0.0", "--port=5000"]

nano .dockerignore
        __pycache__
        *.pyc
        .git

docker images
docker build -t flask-commerce-app .
docker images
docker run -d -p 5000:5000 --name flask-commerce flask-commerce-app
docker ps
docker stop flask-commerce
